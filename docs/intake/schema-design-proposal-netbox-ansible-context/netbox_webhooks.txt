# NetBox Webhook Definitions — Schema Enforcement
# Tenant: cas | Env: lab | Site: hom
# Apply via: NetBox UI → Extras → Webhooks, or NetBox API POST /api/extras/webhooks/

webhooks:

  - name: "schema-enforce-device-create"
    content_types:
      - "dcim.device"
    events:
      - created
    enabled: true
    url: "http://webhook-handler:9000/webhook/netbox/schema-enforce"
    http_method: POST
    http_content_type: "application/json"
    secret: "CHANGE_ME_STRONG_SECRET"
    ssl_verification: true
    conditions:
      attr: "name"
      value: ""
      op: "ne"
    body_template: |
      {
        "event": "{{ event }}",
        "model": "{{ model }}",
        "timestamp": "{{ timestamp }}",
        "data": {
          "id": {{ data['id'] }},
          "name": "{{ data['name'] }}",
          "status": "{{ data['status']['value'] }}",
          "tenant": {% if data['tenant'] %}"{{ data['tenant']['slug'] }}"{% else %}null{% endif %},
          "site": {% if data['site'] %}"{{ data['site']['slug'] }}"{% else %}null{% endif %},
          "device_role": {% if data['device_role'] %}"{{ data['device_role']['slug'] }}"{% else %}null{% endif %}
        }
      }

  - name: "schema-enforce-device-update"
